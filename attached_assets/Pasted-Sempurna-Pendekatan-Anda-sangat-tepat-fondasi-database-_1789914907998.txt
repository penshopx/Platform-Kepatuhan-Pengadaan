```sql
RETURNS TRIGGER AS $$
DECLARE
  v_actor_id UUID;
  v_actor_type VARCHAR(20);
  v_actor_role VARCHAR(50);
  v_reason TEXT;
  v_entity_id TEXT;
BEGIN
  v_actor_id := NULLIF(current_setting('app.actor_id', true), '')::UUID;
  v_actor_type := COALESCE(NULLIF(current_setting('app.actor_type', true), ''), 'SYSTEM');
  v_actor_role := COALESCE(NULLIF(current_setting('app.actor_role', true), ''), 'SYSTEM');

  -- Operasi kritis wajib alasan
  IF TG_OP IN ('UPDATE', 'DELETE') AND v_reason IS NULL THEN
    RAISE EXCEPTION 'Reason is required for UPDATE/DELETE operations on %', TG_TABLE_NAME;
  END IF;

  -- Ambil entity_id dari NEW atau OLD
  IF TG_OP = 'DELETE' THEN
    v_entity_id := OLD.id::TEXT;
  ELSE
    v_entity_id := NEW.id::TEXT;
  END IF;

  IF TG_OP = 'INSERT' THEN
    INSERT INTO audit_logs (actor_id, actor_type, actor_role, action_type, entity_type, entity_id, new_state, reason)
    VALUES (v_actor_id, v_actor_type, v_actor_role, 'CREATE', TG_TABLE_NAME, v_entity_id, to_jsonb(NEW), v_reason);
    RETURN NEW;
  ELSIF TG_OP = 'UPDATE' THEN
    INSERT INTO audit_logs (actor_id, actor_type, actor_role, action_type, entity_type, entity_id, previous_state, new_state, reason)
    VALUES (v_actor_id, v_actor_type, v_actor_role, 'UPDATE', TG_TABLE_NAME, v_entity_id, to_jsonb(OLD), to_jsonb(NEW), v_reason);
    RETURN NEW;
  ELSIF TG_OP = 'DELETE' THEN
    INSERT INTO audit_logs (actor_id, actor_type, actor_role, action_type, entity_type, entity_id, previous_state, reason)
    VALUES (v_actor_id, v_actor_type, v_actor_role, 'DELETE', TG_TABLE_NAME, v_entity_id, to_jsonb(OLD), v_reason);
    RETURN OLD;
  END IF;
END;
$$ LANGUAGE plpgsql;

CREATE OR REPLACE FUNCTION protect_published_policy_pack()
RETURNS TRIGGER AS $$
BEGIN
  -- Jika status lama adalah PUBLISHED
  IF OLD.status = 'PUBLISHED' THEN
    -- Hanya izinkan transisi ke RETIRED
    IF NEW.status != 'RETIRED' THEN
      RAISE EXCEPTION 'Published policy pack can only transition to RETIRED status, not to %', NEW.status;
    END IF;

    -- Pastikan tidak ada perubahan field lain selain status, updated_at
    IF OLD.pack_name != NEW.pack_name OR
       OLD.regime_type != NEW.regime_type OR
       COALESCE(OLD.description, '') != COALESCE(NEW.description, '') OR
       OLD.version != NEW.version OR
       OLD.published_at IS DISTINCT FROM NEW.published_at OR
       OLD.published_by IS DISTINCT FROM NEW.published_by THEN
      RAISE EXCEPTION 'Cannot modify content of published policy pack. Only status can change to RETIRED.';
    END IF;
  END IF;

  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

  v_pack_status VARCHAR(20);

END;
```

```typescript
export class MissingReferenceError extends Error {
  constructor(public path: string) {
    super(`Reference '${path}' not found in context`);
    this.name = 'MissingReferenceError';
  }

export class InvalidOperatorError extends Error {
  constructor(message: string) {

    this.name = 'InvalidOperatorError';

}

  constructor(message: string) {
```

#### `src/rules/ast/evaluator.ts` (Revisi Lengkap)

```typescript
import { AstNode, EvaluationContext, EvaluationResult } from './types';
import { MissingReferenceError, InvalidOperatorError, TypeError } from './errors';

export function evaluateAst(
  ast: AstNode,
  context: EvaluationContext,
  allowedReferences?: string[]
): EvaluationResult {
  // PERBAIKAN #5: Closure internal yang berbagi referencedPaths
  const referencedPaths = new Set<string>();

  function evaluate(node: AstNode): number | string | boolean | null {
    if (node.type === 'literal') {
      return node.value;
    }

    if (node.type === 'reference') {
      if (allowedReferences && !allowedReferences.includes(node.path)) {
        throw new InvalidOperatorError(`Reference '${node.path}' is not allowed`);
      }
      referencedPaths.add(node.path);
      const value = context[node.path];
      if (value === undefined) {
        throw new MissingReferenceError(node.path);
      }
      return value;
    }

    if (node.type === 'operator') {
      return evaluateOperator(node.operator, node.operands);
    }

    if (node.type === 'conditional') {
      const condition = evaluate(node.condition);
      if (typeof condition !== 'boolean') {
        throw new TypeError('Conditional condition must evaluate to boolean');
      }
      // Lazy evaluation: hanya evaluasi branch yang dipilih
      return condition ? evaluate(node.then) : evaluate(node.else);
    }

    throw new InvalidOperatorError('Invalid AST node');
  }

  function evaluateOperator(operator: string, operands: AstNode[]): number | string | boolean | null {
    switch (operator) {
      // PERBAIKAN #4: coalesce hanya tangani MissingReferenceError
      case 'coalesce':
        for (const operand of operands) {
          try {
            const value = evaluate(operand);
            if (value !== null && value !== undefined) {
              return value;
            }
          } catch (e) {
            // Hanya lanjutkan jika error adalah MissingReferenceError
            if (!(e instanceof MissingReferenceError)) {
              throw e; // Re-throw error lain (division by zero, type error, dll)
            }
          }
        }
        return null;

      // PERBAIKAN #5: and/or menggunakan closure internal yang sama
      case 'and':
        for (const operand of operands) {
          const value = evaluate(operand);
          if (typeof value !== 'boolean') {
            throw new TypeError('AND operator requires boolean operands');
          }
          if (!value) return false; // Short-circuit
        }
        return true;

      case 'or':
        for (const operand of operands) {
          const value = evaluate(operand);
          if (typeof value !== 'boolean') {
            throw new TypeError('OR operator requires boolean operands');
          }
          if (value) return true; // Short-circuit
        }
        return false;

      case 'not':
        if (operands.length !== 1) {
          throw new InvalidOperatorError('not requires exactly 1 operand');
        }
        const notValue = evaluate(operands[0]);
        if (typeof notValue !== 'boolean') {
          throw new TypeError('not requires boolean operand');
        }
        return !notValue;

      // Aritmatika
      case 'add':
      case 'subtract':
      case 'multiply':
      case 'divide':
        if (operands.length !== 2) {
          throw new InvalidOperatorError(`${operator} requires exactly 2 operands`);
        }
        const [left, right] = operands.map(op => evaluate(op));
        if (typeof left !== 'number' || typeof right !== 'number') {
          throw new TypeError(`${operator} requires numeric operands`);
        }
        if (operator === 'add') return left + right;
        if (operator === 'subtract') return left - right;
        if (operator === 'multiply') return left * right;
        if (operator === 'divide') {
          if (right === 0) throw new InvalidOperatorError('Division by zero');
          return left / right;
        }
        break;

      // Agregasi
      case 'min':
      case 'max':
        if (operands.length < 1) {
          throw new InvalidOperatorError(`${operator} requires at least 1 operand`);
        }
        const aggValues = operands.map(op => evaluate(op));
        if (!aggValues.every(v => typeof v === 'number')) {
          throw new TypeError(`${operator} requires numeric operands`);
        }
        return operator === 'min' ? Math.min(...aggValues as number[]) : Math.max(...aggValues as number[]);

      // Numerik
      case 'round':
      case 'floor':
      case 'ceil':
      case 'abs':
        if (operands.length !== 1) {
          throw new InvalidOperatorError(`${operator} requires exactly 1 operand`);
        }
        const numValue = evaluate(operands[0]);
        if (typeof numValue !== 'number') {
          throw new TypeError(`${operator} requires numeric operand`);
        }
        if (operator === 'round') return Math.round(numValue);
        if (operator === 'floor') return Math.floor(numValue);
        if (operator === 'ceil') return Math.ceil(numValue);
        if (operator === 'abs') return Math.abs(numValue);
        break;

      case 'clamp':
        if (operands.length !== 3) {
          throw new InvalidOperatorError('clamp requires exactly 3 operands');
        }
        const clampValues = operands.map(op => evaluate(op));
        if (!clampValues.every(v => typeof v === 'number')) {
          throw new TypeError('clamp requires numeric operands');
        }
        return Math.min(Math.max(clampValues[0] as number, clampValues[1] as number), clampValues[2] as number);

      // Perbandingan
      case 'eq':
      case 'neq':
        if (operands.length !== 2) {
          throw new InvalidOperatorError(`${operator} requires exactly 2 operands`);
        }
        const [eqLeft, eqRight] = operands.map(op => evaluate(op));
        return operator === 'eq' ? eqLeft === eqRight : eqLeft !== eqRight;

      case 'gt':
      case 'gte':
      case 'lt':
      case 'lte':
        if (operands.length !== 2) {
          throw new InvalidOperatorError(`${operator} requires exactly 2 operands`);
        }
        const [cmpLeft, cmpRight] = operands.map(op => evaluate(op));
        if (typeof cmpLeft !== 'number' || typeof cmpRight !== 'number') {
          throw new TypeError(`${operator} requires numeric operands`);
        }
        if (operator === 'gt') return cmpLeft > cmpRight;
        if (operator === 'gte') return cmpLeft >= cmpRight;
        if (operator === 'lt') return cmpLeft < cmpRight;
        if (operator === 'lte') return cmpLeft <= cmpRight;
        break;

      default:
        throw new InvalidOperatorError(`Unknown operator: ${operator}`);
    }

    throw new InvalidOperatorError(`Unhandled operator: ${operator}`);
  }

  const result = evaluate(ast);

  return {
    value: result,
    referencedPaths: Array.from(referencedPaths),
    sourceVerificationStatus: 'VERIFIED' // Akan di-set oleh evaluator level tinggi
  };
}
```
