const handleRetry = async () => {
  setError(null);
  setLoading(true);
  try {
    const packs = await apiClient.getPublishedPolicyPacks();
    setPolicyPacks(packs);
  } catch (e) {
    setError(e instanceof Error ? e.message : 'Unknown error');
  } finally {
    setLoading(false);
  }
};
