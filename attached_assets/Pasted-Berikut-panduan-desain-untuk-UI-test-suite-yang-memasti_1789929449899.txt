## 🧪 STRATEGI TESTING

### Mengapa Mock API Client (Bukan MSW/HTTP)

1. **Isolasi**: Test tidak bergantung pada API server yang berjalan
2. **Performa**: Lebih cepat karena tidak ada network overhead
3. **Deterministik**: Response selalu sama, tidak ada flakiness
4. **Fokus**: Menguji perilaku komponen, bukan integrasi HTTP

## 🔧 MOCK API CLIENT

### `mocks/api-client.ts`

```typescript
import { vi } from 'vitest';
import type { EvaluationResponse, PolicyPackSummary } from '@/lib/api-client-react/src/generated/api';

export const mockApiClient = {
  getPublishedPolicyPacks: vi.fn(),
  postRuleEvaluations: vi.fn()
};

export function setupMockApiResponses(scenario: {
  policyPacks?: PolicyPackSummary[];
  evaluation?: EvaluationResponse | null;
  evaluationError?: Error;
}) {
  if (scenario.policyPacks !== undefined) {
    mockApiClient.getPublishedPolicyPacks.mockResolvedValue(scenario.policyPacks);
  }

  if (scenario.evaluation !== undefined) {
    if (scenario.evaluation === null) {
      mockApiClient.postRuleEvaluations.mockResolvedValue(null);
    } else {
      mockApiClient.postRuleEvaluations.mockResolvedValue(scenario.evaluation);
    }
  }

  if (scenario.evaluationError) {
    mockApiClient.postRuleEvaluations.mockRejectedValue(scenario.evaluationError);
  }
}

export function resetMockApi() {
  mockApiClient.getPublishedPolicyPacks.mockReset();
  mockApiClient.postRuleEvaluations.mockReset();
}
```

import { describe, it, expect, beforeEach, vi } from 'vitest';
import { render, screen, waitFor, fireEvent } from '@testing-library/react';
import userEvent from '@testing-library/user-event';
import { PackageComplianceEvaluation } from '../package-compliance-evaluation';
import { mockApiClient, setupMockApiResponses, resetMockApi } from './mocks/api-client';
import { 
  publishedPolicyPacks, 
  emptyPolicyPacks, 
  publishedPolicyPack 
} from './fixtures/policy-packs';
import { 
  successfulEvaluation, 
  failedEvaluation, 
  evaluationWithOverride 
} from './fixtures/evaluations';
import { packageContext, packageContextFailed } from './fixtures/contexts';

// Mock API client
vi.mock('@/lib/api-client', () => ({
  apiClient: mockApiClient
}));

describe('PackageComplianceEvaluation', () => {
  beforeEach(() => {
    resetMockApi();
  });

  describe('Initial State', () => {
    it('should show loading state while fetching policy packs', async () => {
      setupMockApiResponses({
        policyPacks: new Promise(() => {}) // Never resolves
      });

      render(<PackageComplianceEvaluation packageContext={packageContext} />);

      expect(screen.getByText(/memuat policy pack/i)).toBeInTheDocument();
    });

    it('should show empty state when no published policy packs', async () => {
      setupMockApiResponses({
        policyPacks: emptyPolicyPacks
      });

      render(<PackageComplianceEvaluation packageContext={packageContext} />);

      await waitFor(() => {
        expect(screen.getByText(/tidak ada policy pack tersedia/i)).toBeInTheDocument();
      });

      expect(screen.queryByRole('button', { name: /mulai evaluasi/i })).toBeDisabled();
    });

    it('should show policy pack selector when packs are available', async () => {
      setupMockApiResponses({
        policyPacks: publishedPolicyPacks
      });

      render(<PackageComplianceEvaluation packageContext={packageContext} />);

      await waitFor(() => {
        expect(screen.getByRole('combobox', { name: /pilih policy pack/i })).toBeInTheDocument();
      });
    });
  });

  describe('Successful Evaluation', () => {
    beforeEach(() => {
      setupMockApiResponses({
        policyPacks: publishedPolicyPacks,
        evaluation: successfulEvaluation
      });
    });

    it('should display evaluation summary with correct counts', async () => {
      const user = userEvent.setup();
      render(<PackageComplianceEvaluation packageContext={packageContext} />);

      // Select policy pack
      await waitFor(() => {
        expect(screen.getByRole('combobox')).toBeInTheDocument();
      });
      await user.click(screen.getByRole('combobox'));
      await user.click(screen.getByText(publishedPolicyPack.packName));

      // Click evaluate
      await user.click(screen.getByRole('button', { name: /mulai evaluasi/i }));

      // Wait for results
      await waitFor(() => {
        expect(screen.getByText(/1 lulus/i)).toBeInTheDocument();
        expect(screen.getByText(/1 tidak berlaku/i)).toBeInTheDocument();
      });
    });

    it('should display decision reference with copy button', async () => {
      const user = userEvent.setup();
      render(<PackageComplianceEvaluation packageContext={packageContext} />);

      await waitFor(() => {
        expect(screen.getByRole('combobox')).toBeInTheDocument();
      });
      await user.click(screen.getByRole('combobox'));
      await user.click(screen.getByText(publishedPolicyPack.packName));
      await user.click(screen.getByRole('button', { name: /mulai evaluasi/i }));

      await waitFor(() => {
        expect(screen.getByText(/decision reference/i)).toBeInTheDocument();
        expect(screen.getByText(successfulEvaluation.decisionReferenceId)).toBeInTheDocument();
      });

      // Test copy button
      const copyButton = screen.getByRole('button', { name: /salin decision reference/i });
      await user.click(copyButton);

      // Verify clipboard (mock)
      expect(navigator.clipboard.writeText).toHaveBeenCalledWith(
        successfulEvaluation.decisionReferenceId
      );
    });

    it('should display rule explanation with all required fields', async () => {
      const user = userEvent.setup();
      render(<PackageComplianceEvaluation packageContext={packageContext} />);

      await waitFor(() => {
        expect(screen.getByRole('combobox')).toBeInTheDocument();
      });
      await user.click(screen.getByRole('combobox'));
      await user.click(screen.getByText(publishedPolicyPack.packName));
      await user.click(screen.getByRole('button', { name: /mulai evaluasi/i }));

      await waitFor(() => {
        expect(screen.getByText(/tkdn_min_threshold_construction/i)).toBeInTheDocument();
      });

      // Expand rule detail
      const detailButton = screen.getByRole('button', { name: /detail/i });
      await user.click(detailButton);

      // Verify all required fields are displayed
      expect(screen.getByText(/pasal: 62/i)).toBeInTheDocument();
      expect(screen.getByText(/sumber: PERPRES-46-2025/i)).toBeInTheDocument();
      expect(screen.getByText(/status verifikasi/i)).toBeInTheDocument();
      expect(screen.getByText(/VERIFIED/i)).toBeInTheDocument();
      expect(screen.getByText(/mengikat: ya/i)).toBeInTheDocument();
      expect(screen.getByText(/tkdn minimal: 40%/i)).toBeInTheDocument();
      expect(screen.getByText(/tkdn aktual: 45%/i)).toBeInTheDocument();
    });

        expect(screen.getByRole('combobox')).toBeInTheDocument();
      });
      await user.click(screen.getByRole('combobox'));
      await user.click(screen.getByText(publishedPolicyPack.packName));
      await user.click(screen.getByRole('button', { name: /mulai evaluasi/i }));

      await waitFor(() => {
        expect(screen.getByText(/tkdn_min_threshold_construction/i)).toBeInTheDocument();

      expect(screen.getByText(/tkdn_percentage/i)).toBeInTheDocument();

  describe('Failed Evaluation', () => {
    beforeEach(() => {
      setupMockApiResponses({
        policyPacks: publishedPolicyPacks,

      const user = userEvent.setup();

      await waitFor(() => {

      await user.click(screen.getByRole('combobox'));
      await user.click(screen.getByText(publishedPolicyPack.packName));
      await user.click(screen.getByRole('button', { name: /mulai evaluasi/i }));

      await waitFor(() => {
        expect(screen.getByText(/1 gagal/i)).toBeInTheDocument();
      });

      // Expand rule detail
      const detailButton = screen.getByRole('button', { name: /detail/i });
      await user.click(detailButton);

      // Verify failed explanation
      expect(screen.getByText(/tidak terpenuhi/i)).toBeInTheDocument();
      expect(screen.getByText(/tkdn minimal: 40%/i)).toBeInTheDocument();
      expect(screen.getByText(/tkdn aktual: 35%/i)).toBeInTheDocument();

  describe('Evaluation with Override', () => {
    beforeEach(() => {
      setupMockApiResponses({
        policyPacks: publishedPolicyPacks,

      await user.click(screen.getByText(publishedPolicyPack.packName));
      await user.click(screen.getByRole('button', { name: /mulai evaluasi/i }));

      await waitFor(() => {
        expect(screen.getByText(/override diterapkan/i)).toBeInTheDocument();
      });

      const detailButton = screen.getByRole('button', { name: /detail/i });
      await user.click(detailButton);

      expect(screen.getByText(/override diterapkan/i)).toBeInTheDocument();
      expect(screen.getByText(/REGENCY/i)).toBeInTheDocument();
      expect(screen.getByText(/JAKARTA/i)).toBeInTheDocument();

  describe('Error States', () => {
    it('should display error message when evaluation fails', async () => {
      const user = userEvent.setup();
      setupMockApiResponses({
        policyPacks: publishedPolicyPacks,
        evaluationError: new Error('Konteks tidak lengkap: field "tkdn_percentage" wajib diisi')

      render(<PackageComplianceEvaluation packageContext={packageContext} />);

      await waitFor(() => {
        expect(screen.getByRole('combobox')).toBeInTheDocument();

      await waitFor(() => {
        expect(screen.getByText(/evaluasi gagal/i)).toBeInTheDocument();
        expect(screen.getByText(/konteks tidak lengkap/i)).toBeInTheDocument();

      expect(screen.getByRole('button', { name: /coba lagi/i })).toBeInTheDocument();

    it('should allow retry after error', async () => {

        expect(screen.getByText(/evaluasi gagal/i)).toBeInTheDocument();
      });

      // Click retry
      await user.click(screen.getByRole('button', { name: /coba lagi/i }));

      // Should show results after retry
      await waitFor(() => {
        expect(screen.getByText(/1 lulus/i)).toBeInTheDocument();
      });
    });
  });

  describe('Accessibility', () => {
    it('should have proper ARIA labels', async () => {
      setupMockApiResponses({
        policyPacks: publishedPolicyPacks,
        evaluation: successfulEvaluation

      render(<PackageComplianceEvaluation packageContext={packageContext} />);

      await waitFor(() => {

      });

      expect(screen.getByRole('button', { name: /mulai evaluasi/i })).toBeInTheDocument();
    });
  });
});

## ⚠️ CATATAN PENTING

1. **Mock navigator.clipboard**: Tambahkan setup di `vitest.config.ts` atau `setupTests.ts`:
```typescript
Object.assign(navigator, {
  clipboard: {
    writeText: vi.fn()
  }
});
```

2. **Component props**: Sesuaikan dengan signature aktual `PackageComplianceEvaluation`. Jika komponen menerima props lain (misal: `packageId`), tambahkan ke fixture.

3. **API client path**: Sesuaikan path import `@/lib/api-client` dengan struktur aktual workspace.

4. **Testing library**: Pastikan `@testing-library/react`, `@testing-library/user-event`, dan `vitest` sudah terinstall.

5. **Invariant yang dijaga**:
   - Explanation selalu terstruktur (template + data), bukan HTML mentah
   - Decision reference ID ditampilkan persis seperti dari API
   - Semua field wajib (pasal, sumber, verifikasi, input, output) harus tampil
   - Loading, error, empty state ditangani dengan jelas

Silakan lanjutkan implementasi. Jika ada inkonsistensi antara fixture dan response API aktual, saya siap membantu menyesuaikan.
