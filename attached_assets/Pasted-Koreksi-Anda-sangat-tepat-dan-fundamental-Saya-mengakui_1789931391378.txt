## 🔍 ANALISIS MASALAH

### Mengapa Contoh Sebelumnya Tidak Bekerja

```typescript
// ❌ SALAH: Ini tidak akan bekerja
beforeEach(async () => {
  testDb = await beginTransaction(); // Koneksi A, transaksi dimulai
  
  // Data dibuat di koneksi A, belum commit
  const [dec1] = await testDb.insert(decisionReferences).values({...}).returning();
});

it('should return decision', async () => {
  // Supertest menggunakan koneksi B dari pool aplikasi
  // Koneksi B TIDAK bisa melihat data di koneksi A yang belum commit!
  const res = await request(app)
    .get(`/api/decisions/${decisionId1}`)
    .set(testAuthHeader(TEST_USERS.ADMIN));
  
  // res.status akan 404 karena data tidak terlihat
});
```

**Masalahnya:**
- PostgreSQL default isolation level: `READ COMMITTED`
- Transaksi di koneksi A tidak terlihat oleh koneksi B sampai `COMMIT`
- Supertest menggunakan connection pool aplikasi, bukan `testDb`
- Test akan gagal dengan 404 atau data tidak konsisten

## 🔐 AUTHORIZATION: Mock Clerk Verifier

### Implementasi yang Benar

```typescript
// artifacts/api-server/src/middleware/clerk.ts
import { getAuth } from '@clerk/express';
import { verifyToken } from '@clerk/backend';

export function clerkAuthMiddleware(req, res, next) {
  if (process.env.NODE_ENV === 'test') {
    return testClerkAuth(req, res, next);
  }
  return productionClerkAuth(req, res, next);
}

async function productionClerkAuth(req, res, next) {
  try {
    const token = req.headers.authorization?.replace('Bearer ', '');
    if (!token) {
      return res.status(401).json({ error: 'Token required' });
    }
    
    // Verifikasi token dengan Clerk asli
    const payload = await verifyToken(token, { 
      secretKey: process.env.CLERK_SECRET_KEY 
    });
    
    req.auth = { userId: payload.sub };
    next();
  } catch (error) {
    return res.status(401).json({ error: 'Invalid token' });
  }
}

async function testClerkAuth(req, res, next) {
  // Test environment: gunakan mock verifier
  const testUserId = req.headers['x-test-user-id'];
  if (!testUserId) {
    return res.status(401).json({ error: 'X-Test-User-Id required in test environment' });
  }
  
  req.auth = { userId: testUserId };
  next();
}
```

**Keuntungan:**
- Aman di production (header `X-Test-User-Id` diabaikan)
- Eksplisit (developer tahu test pakai mock)
- Tidak ada backdoor
