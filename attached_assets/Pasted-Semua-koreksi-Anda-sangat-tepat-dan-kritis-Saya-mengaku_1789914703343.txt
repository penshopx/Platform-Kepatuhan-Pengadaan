### 3. EVALUATOR AST YANG SUDAH DIPERBAIKI (LAZY EVALUATION)

#### `src/rules/ast/evaluator.ts`

```typescript
import { AstNode, EvaluationContext, EvaluationResult } from './types';

export function evaluateAst(
  ast: AstNode,
  context: EvaluationContext,
  allowedReferences?: string[]
): EvaluationResult {
  const referencedPaths = new Set<string>();

  // CORRECTION: Lazy evaluation function
  function evaluate(node: AstNode): number | string | boolean | null {
    if (node.type === 'literal') {
      return node.value;
    }

    if (node.type === 'reference') {
      if (allowedReferences && !allowedReferences.includes(node.path)) {
        throw new Error(`Reference '${node.path}' is not allowed`);
      }
      referencedPaths.add(node.path);
      const value = context[node.path];
      if (value === undefined) {
        throw new Error(`Reference '${node.path}' not found in context`);
      }
      return value;
    }

    if (node.type === 'operator') {
      // CORRECTION: Lazy evaluation untuk operator tertentu
      return evaluateOperatorLazy(node.operator, node.operands, context, allowedReferences);
    }

    if (node.type === 'conditional') {
      // CORRECTION: Lazy evaluation untuk conditional
      const condition = evaluate(node.condition);
      if (typeof condition !== 'boolean') {
        throw new Error('Conditional condition must evaluate to boolean');
      }
      return condition ? evaluate(node.then) : evaluate(node.else);
    }

    throw new Error('Invalid AST node');
  }

  const result = evaluate(ast);

  return {
    value: result,
    referencedPaths: Array.from(referencedPaths),
    sourceVerificationStatus: 'VERIFIED' // Akan di-set oleh evaluator level tinggi
  };
}

// CORRECTION: Lazy evaluation untuk operator tertentu
function evaluateOperatorLazy(
  operator: string,
  operands: AstNode[],
  context: EvaluationContext,
  allowedReferences?: string[]
): number | string | boolean | null {
  switch (operator) {
    // CORRECTION: coalesce - evaluasi satu per satu, berhenti saat menemukan non-null
    case 'coalesce':
      for (const operand of operands) {
        try {
          const value = evaluateAst(operand, context, allowedReferences).value;
          if (value !== null && value !== undefined) {
            return value;
          }
        } catch (e) {
          // Lanjut ke operand berikutnya jika referensi tidak ditemukan
          continue;
        }
      }
      return null;

    // CORRECTION: and - short-circuit evaluation
    case 'and':
      for (const operand of operands) {
        const value = evaluateAst(operand, context, allowedReferences).value;
        if (typeof value !== 'boolean') {
          throw new Error('AND operator requires boolean operands');
        }
        if (!value) return false; // Short-circuit
      }
      return true;

    // CORRECTION: or - short-circuit evaluation
    case 'or':
      for (const operand of operands) {
        const value = evaluateAst(operand, context, allowedReferences).value;
        if (typeof value !== 'boolean') {
          throw new Error('OR operator requires boolean operands');
        }
        if (value) return true; // Short-circuit
      }
      return false;

    // Operator lain - evaluasi semua operand dulu
    default: {
      const evaluatedOperands = operands.map(op => evaluateAst(op, context, allowedReferences).value);
      return evaluateOperator(operator, evaluatedOperands);
    }
  }
}

function evaluateOperator(operator: string, operands: any[]): number | string | boolean {
  // CORRECTION: Validasi arity dan tipe
  switch (operator) {
    // Aritmatika - butuh 2 operand numerik
    case 'add':
    case 'subtract':
    case 'multiply':
    case 'divide':
      if (operands.length !== 2) {
        throw new Error(`${operator} requires exactly 2 operands`);
      }
      if (typeof operands[0] !== 'number' || typeof operands[1] !== 'number') {
        throw new Error(`${operator} requires numeric operands`);
      }
      if (operator === 'add') return operands[0] + operands[1];
      if (operator === 'subtract') return operands[0] - operands[1];
      if (operator === 'multiply') return operands[0] * operands[1];
      if (operator === 'divide') {
        if (operands[1] === 0) throw new Error('Division by zero');
        return operands[0] / operands[1];
      }
      break;

    // Agregasi - butuh minimal 1 operand
    case 'min':
    case 'max':
      if (operands.length < 1) {
        throw new Error(`${operator} requires at least 1 operand`);
      }
      if (!operands.every(op => typeof op === 'number')) {
        throw new Error(`${operator} requires numeric operands`);
      }
      return operator === 'min' ? Math.min(...operands) : Math.max(...operands);

    // Numerik - butuh 1 operand numerik
    case 'round':
    case 'floor':
    case 'ceil':
    case 'abs':
      if (operands.length !== 1) {
        throw new Error(`${operator} requires exactly 1 operand`);
      }
      if (typeof operands[0] !== 'number') {
        throw new Error(`${operator} requires numeric operand`);
      }
      if (operator === 'round') return Math.round(operands[0]);
      if (operator === 'floor') return Math.floor(operands[0]);
      if (operator === 'ceil') return Math.ceil(operands[0]);
      if (operator === 'abs') return Math.abs(operands[0]);
      break;

    // Clamp - butuh 3 operand
    case 'clamp':
      if (operands.length !== 3) {
        throw new Error('clamp requires exactly 3 operands');
      }
      if (!operands.every(op => typeof op === 'number')) {
        throw new Error('clamp requires numeric operands');
      }
      return Math.min(Math.max(operands[0], operands[1]), operands[2]);

    // Perbandingan - butuh 2 operand
    case 'eq':
    case 'neq':
      if (operands.length !== 2) {
        throw new Error(`${operator} requires exactly 2 operands`);
      }
      return operator === 'eq' ? operands[0] === operands[1] : operands[0] !== operands[1];

    case 'gt':
    case 'gte':
    case 'lt':
    case 'lte':
      if (operands.length !== 2) {
        throw new Error(`${operator} requires exactly 2 operands`);
      }
      if (typeof operands[0] !== 'number' || typeof operands[1] !== 'number') {
        throw new Error(`${operator} requires numeric operands`);
      }
      if (operator === 'gt') return operands[0] > operands[1];
      if (operator === 'gte') return operands[0] >= operands[1];
      if (operator === 'lt') return operands[0] < operands[1];
      if (operator === 'lte') return operands[0] <= operands[1];
      break;

    // Not - butuh 1 operand boolean
    case 'not':
      if (operands.length !== 1) {
        throw new Error('not requires exactly 1 operand');
      }
      if (typeof operands[0] !== 'boolean') {
        throw new Error('not requires boolean operand');
      }
      return !operands[0];

    default:
      throw new Error(`Unknown operator: ${operator}`);
  }

  throw new Error(`Unhandled operator: ${operator}`);
}
```

### 4. CANONICAL JSON SERIALIZER

#### `src/utils/canonicalJson.ts`

```typescript
/**
 * CORRECTION: Serialisasi JSON kanonik untuk hash yang konsisten
 * Mengurutkan key secara rekursif dan menangani tipe data khusus
 */
export function canonicalJsonStringify(obj: any): string {
  if (obj === null || obj === undefined) {
    return 'null';
  }

  if (typeof obj === 'number' || typeof obj === 'boolean') {
    return JSON.stringify(obj);
  }

  if (typeof obj === 'string') {
    return JSON.stringify(obj);
  }

  if (Array.isArray(obj)) {
    const items = obj.map(item => canonicalJsonStringify(item));
    return `[${items.join(',')}]`;
  }

  if (typeof obj === 'object') {
    const keys = Object.keys(obj).sort();
    const pairs = keys.map(key => {
      const value = canonicalJsonStringify(obj[key]);
      return `${JSON.stringify(key)}:${value}`;
    });
    return `{${pairs.join(',')}}`;
  }

  throw new Error(`Cannot serialize type: ${typeof obj}`);
}
```

```typescript
import { canonicalJsonStringify } from '@/utils/canonicalJson';
import crypto from 'crypto';

function generateSnapshotHash(rule: any, condition: any, source: any): string {
  // CORRECTION: Gunakan canonical JSON serialization
  const data = canonicalJsonStringify({
    ruleId: rule.id,
    ruleKey: rule.ruleKey,
    version: rule.version,
    defaultValue: rule.defaultValue,
    formulaAst: rule.formulaAst,
    conditionId: condition?.id || null,
    sourceId: source.id,
    provisionId: rule.provisionId
  });

  return crypto.createHash('sha256').update(data).digest('hex');
}
```
