```sql
CREATE OR REPLACE FUNCTION prevent_audit_log_modification()

BEGIN

-- Trigger untuk auto-log perubahan pada tabel kritis
CREATE OR REPLACE FUNCTION log_table_changes()
RETURNS TRIGGER AS $$
DECLARE
  v_actor_id UUID;
  v_actor_type VARCHAR(20);
  v_reason TEXT;
BEGIN
  -- Ambil actor dari session variable (diset oleh aplikasi)
  v_actor_id := current_setting('app.actor_id', true)::UUID;
  v_actor_type := COALESCE(current_setting('app.actor_type', true), 'SYSTEM');
  v_reason := current_setting('app.action_reason', true);

  IF TG_OP = 'INSERT' THEN
    INSERT INTO audit_logs (actor_id, actor_type, actor_role, action_type, entity_type, entity_id, new_state, reason)
    VALUES (v_actor_id, v_actor_type, 'SYSTEM', 'CREATE', TG_TABLE_NAME, NEW.id, to_jsonb(NEW), v_reason);
    RETURN NEW;
  ELSIF TG_OP = 'UPDATE' THEN
    INSERT INTO audit_logs (actor_id, actor_type, actor_role, action_type, entity_type, entity_id, previous_state, new_state, reason)
    VALUES (v_actor_id, v_actor_type, 'SYSTEM', 'UPDATE', TG_TABLE_NAME, NEW.id, to_jsonb(OLD), to_jsonb(NEW), v_reason);
    RETURN NEW;
  ELSIF TG_OP = 'DELETE' THEN
    INSERT INTO audit_logs (actor_id, actor_type, actor_role, action_type, entity_type, entity_id, previous_state, reason)
    VALUES (v_actor_id, v_actor_type, 'SYSTEM', 'DELETE', TG_TABLE_NAME, OLD.id, to_jsonb(OLD), v_reason);
    RETURN OLD;
  END IF;
END;
$$ LANGUAGE plpgsql;
```

```typescript
export type AstNode = LiteralNode | ReferenceNode | OperatorNode | ConditionalNode;

export interface LiteralNode {
  type: 'literal';
  value: number | string | boolean;
}

export interface ReferenceNode {
  type: 'reference';
  path: string; // hanya alfanumerik dan underscore
}

export interface OperatorNode {
  type: 'operator';
  operator: 'add' | 'subtract' | 'multiply' | 'divide' | 'min' | 'max' | 
            'round' | 'floor' | 'ceil' | 'abs' | 'clamp' |
            'eq' | 'neq' | 'gt' | 'gte' | 'lt' | 'lte' |
            'and' | 'or' | 'not' | 'coalesce';
  operands: AstNode[];
}

export interface ConditionalNode {
  type: 'conditional';
  condition: AstNode; // harus boolean
  then: AstNode;
  else: AstNode;
}

export interface EvaluationContext {
  [key: string]: number | string | boolean | null;
}

export interface EvaluationResult {
  value: number | string | boolean;
  referencedPaths: string[];
  sourceVerificationStatus: 'VERIFIED' | 'UNVERIFIED' | 'NON_BINDING';
}
```

#### `src/rules/ast/validator.ts`

```typescript
import { z } from 'zod';
import { AstNode } from './types';

const MAX_DEPTH = 10;
const MAX_NODES = 100;
const ALLOWED_OPERATORS = [
  'add', 'subtract', 'multiply', 'divide', 'min', 'max',
  'round', 'floor', 'ceil', 'abs', 'clamp',
  'eq', 'neq', 'gt', 'gte', 'lt', 'lte',
  'and', 'or', 'not', 'coalesce'
] as const;

const LiteralNodeSchema: z.ZodType<any> = z.object({
  type: z.literal('literal'),
  value: z.union([z.number(), z.string(), z.boolean()])
});

const ReferenceNodeSchema: z.ZodType<any> = z.object({
  type: z.literal('reference'),
  path: z.string().regex(/^[a-zA-Z0-9_.]+$/, 'Path must be alphanumeric with underscores')
});

const OperatorNodeSchema: z.ZodType<any> = z.lazy(() =>
  z.object({
    type: z.literal('operator'),
    operator: z.enum(ALLOWED_OPERATORS),
    operands: z.array(AstNodeSchema).min(1)
  })
);

const ConditionalNodeSchema: z.ZodType<any> = z.lazy(() =>
  z.object({
    type: z.literal('conditional'),
    condition: AstNodeSchema,
    then: AstNodeSchema,
    else: AstNodeSchema
  })
);

const AstNodeSchema: z.ZodType<any> = z.lazy(() =>
  z.union([LiteralNodeSchema, ReferenceNodeSchema, OperatorNodeSchema, ConditionalNodeSchema])
);

export function validateAst(ast: any, allowedReferences?: string[]): { valid: boolean; error?: string } {
  // Validasi struktur dasar
  const parseResult = AstNodeSchema.safeParse(ast);
  if (!parseResult.success) {
    return { valid: false, error: `Invalid AST structure: ${parseResult.error.message}` };
  }

  // Validasi kedalaman dan jumlah node
  const stats = calculateAstStats(ast);
  if (stats.depth > MAX_DEPTH) {
    return { valid: false, error: `AST depth ${stats.depth} exceeds maximum ${MAX_DEPTH}` };
  }
  if (stats.nodeCount > MAX_NODES) {
    return { valid: false, error: `AST node count ${stats.nodeCount} exceeds maximum ${MAX_NODES}` };
  }

  // Validasi referensi yang diizinkan
  if (allowedReferences) {
    const refs = extractReferences(ast);
    for (const ref of refs) {
      if (!allowedReferences.includes(ref)) {
        return { valid: false, error: `Reference '${ref}' is not allowed` };
      }
    }
  }

  return { valid: true };
}

function calculateAstStats(node: any, depth: number = 0): { depth: number; nodeCount: number } {
  let maxDepth = depth;
  let nodeCount = 1;

  if (node.type === 'operator') {
    for (const operand of node.operands) {
      const stats = calculateAstStats(operand, depth + 1);
      maxDepth = Math.max(maxDepth, stats.depth);
      nodeCount += stats.nodeCount;
    }
  } else if (node.type === 'conditional') {
    const condStats = calculateAstStats(node.condition, depth + 1);
    const thenStats = calculateAstStats(node.then, depth + 1);
    const elseStats = calculateAstStats(node.else, depth + 1);
    maxDepth = Math.max(maxDepth, condStats.depth, thenStats.depth, elseStats.depth);
    nodeCount += condStats.nodeCount + thenStats.nodeCount + elseStats.nodeCount;
  }

  return { depth: maxDepth, nodeCount };
}

function extractReferences(node: any): string[] {
  const refs: string[] = [];
  
  if (node.type === 'reference') {
    refs.push(node.path);
  } else if (node.type === 'operator') {
    for (const operand of node.operands) {
      refs.push(...extractReferences(operand));
    }
  } else if (node.type === 'conditional') {
    refs.push(...extractReferences(node.condition));
    refs.push(...extractReferences(node.then));
    refs.push(...extractReferences(node.else));
  }

  return refs;
}
```

#### `src/rules/ast/evaluator.ts`

```typescript
import { AstNode, EvaluationContext, EvaluationResult } from './types';

export function evaluateAst(
  ast: AstNode, 
  context: EvaluationContext,
  allowedReferences?: string[]
): EvaluationResult {
  const referencedPaths = new Set<string>();
  
  function evaluate(node: AstNode): number | string | boolean {
    if (node.type === 'literal') {
      return node.value;
    }

    if (node.type === 'reference') {
      if (allowedReferences && !allowedReferences.includes(node.path)) {
        throw new Error(`Reference '${node.path}' is not allowed`);
      }
      referencedPaths.add(node.path);
      const value = context[node.path];
      if (value === undefined || value === null) {
        throw new Error(`Reference '${node.path}' not found in context`);
      }
      return value;
    }

    if (node.type === 'operator') {
      const operands = node.operands.map(op => evaluate(op));
      return evaluateOperator(node.operator, operands);
    }

    if (node.type === 'conditional') {
      const condition = evaluate(node.condition);
      if (typeof condition !== 'boolean') {
        throw new Error('Conditional condition must evaluate to boolean');
      }
      return condition ? evaluate(node.then) : evaluate(node.else);
    }

    throw new Error('Invalid AST node');
  }

  const result = evaluate(ast);
  
  return {
    value: result,
    referencedPaths: Array.from(referencedPaths),
    sourceVerificationStatus: 'VERIFIED' // Akan di-set oleh evaluator level tinggi
  };
}

function evaluateOperator(operator: string, operands: any[]): number | string | boolean {
  switch (operator) {
    // Aritmatika
    case 'add': return (operands[0] as number) + (operands[1] as number);
    case 'subtract': return (operands[0] as number) - (operands[1] as number);
    case 'multiply': return (operands[0] as number) * (operands[1] as number);
    case 'divide':
      if (operands[1] === 0) throw new Error('Division by zero');
      return (operands[0] as number) / (operands[1] as number);
    
    // Agregasi
    case 'min': return Math.min(...operands as number[]);
    case 'max': return Math.max(...operands as number[]);
    
    // Numerik
    case 'round': return Math.round(operands[0] as number);
    case 'floor': return Math.floor(operands[0] as number);
    case 'ceil': return Math.ceil(operands[0] as number);
    case 'abs': return Math.abs(operands[0] as number);
    case 'clamp': {
      const value = operands[0] as number;
      const min = operands[1] as number;
      const max = operands[2] as number;
      return Math.min(Math.max(value, min), max);
    }
    
    // Perbandingan
    case 'eq': return operands[0] === operands[1];
    case 'neq': return operands[0] !== operands[1];
    case 'gt': return (operands[0] as number) > (operands[1] as number);
    case 'gte': return (operands[0] as number) >= (operands[1] as number);
    case 'lt': return (operands[0] as number) < (operands[1] as number);
    case 'lte': return (operands[0] as number) <= (operands[1] as number);
    
    // Logika
    case 'and': return operands.every(op => op as boolean);
    case 'or': return operands.some(op => op as boolean);
    case 'not': return !(operands[0] as boolean);
    
    // Null handling
    case 'coalesce':
      for (const op of operands) {
        if (op !== null && op !== undefined) return op;
      }
      return null;
    
    default:
      throw new Error(`Unknown operator: ${operator}`);
  }
}
```

```typescript
import { db } from '@/db';
import { policyPacks, policyPackRules, businessRules, ruleConditions, legalSources, provisions } from '@/db/schema';
import { eq, and } from 'drizzle-orm';
import crypto from 'crypto';

interface PublishResult {
  success: boolean;
  error?: string;
  policyPackId?: string;
}

export async function publishPolicyPack(
  policyPackId: string,
  publishedBy: string
): Promise<PublishResult> {
  return await db.transaction(async (tx) => {
    // 1. Ambil policy pack
    const [pack] = await tx.select().from(policyPacks).where(eq(policyPacks.id, policyPackId));
    
    if (!pack) {
      return { success: false, error: 'Policy pack not found' };
    }

    if (pack.status === 'PUBLISHED') {
      return { success: false, error: 'Policy pack already published' };
    }

    // 2. Ambil semua aturan dalam pack
    const packRules = await tx.select().from(policyPackRules).where(eq(policyPackRules.policyPackId, policyPackId));

    // 3. Validasi: tidak boleh ada aturan mengikat yang UNVERIFIED
    for (const packRule of packRules) {
      if (!packRule.isMandatory) continue;

      const [rule] = await tx.select().from(businessRules).where(eq(businessRules.id, packRule.businessRuleId));
      const [provision] = await tx.select().from(provisions).where(eq(provisions.id, rule.provisionId));
      const [source] = await tx.select().from(legalSources).where(eq(legalSources.id, provision.legalSourceId));

      if (source.isBinding && source.verificationStatus === 'UNVERIFIED') {
        return { 
          success: false, 
          error: `Cannot publish: Rule ${rule.ruleKey} references unverified binding source ${source.regulationCode}` 
        };
      }
    }

    // 4. Generate snapshot untuk setiap aturan
    const now = new Date();
    for (const packRule of packRules) {
      const [rule] = await tx.select().from(businessRules).where(eq(businessRules.id, packRule.businessRuleId));
      
      let condition = null;
      if (packRule.ruleConditionId) {
        [condition] = await tx.select().from(ruleConditions).where(eq(ruleConditions.id, packRule.ruleConditionId));
      }

      const [provision] = await tx.select().from(provisions).where(eq(provisions.id, rule.provisionId));
      const [source] = await tx.select().from(legalSources).where(eq(legalSources.id, provision.legalSourceId));

      const snapshot = {
        conditions: condition,
        value: rule.defaultValue,
        formulaAst: rule.formulaAst,
        source: {
          id: source.id,
          regulationCode: source.regulationCode,
          verificationStatus: source.verificationStatus,
          isBinding: source.isBinding
        },
        provision: {
          id: provision.id,
          articleNumber: provision.articleNumber,
          clauseType: provision.clauseType
        },
        hash: generateSnapshotHash(rule, condition, source)
      };

      await tx.update(policyPackRules)
        .set({ 
          snapshot,
          snapshotTakenAt: now
        })
        .where(eq(policyPackRules.id, packRule.id));
    }

    // 5. Update status policy pack
    await tx.update(policyPacks)
      .set({
        status: 'PUBLISHED',
        publishedAt: now,
        publishedBy,
        updatedAt: now
      })
      .where(eq(policyPacks.id, policyPackId));

    return { success: true, policyPackId };
  });
}

function generateSnapshotHash(rule: any, condition: any, source: any): string {
  const data = JSON.stringify({
    ruleId: rule.id,
    ruleKey: rule.ruleKey,
    version: rule.version,
    defaultValue: rule.defaultValue,
    formulaAst: rule.formulaAst,
    conditionId: condition?.id,
    sourceId: source.id,
    provisionId: rule.provisionId
  });
  
  return crypto.createHash('sha256').update(data).digest('hex');
}
```

## 🎯 KESIMPULAN

Implementasi ini mencakup:

✅ **Drizzle sebagai sumber utama** untuk skema, relasi, dan tipe  
✅ **SQL manual** untuk trigger append-only dan constraint audit  
✅ **Copy-on-publish** dengan snapshot terstruktur lengkap + hash integritas  
✅ **AST bertipe ketat** dengan validator Zod dan evaluator aman  
✅ **Operator terbatas** termasuk conditional node khusus  
✅ **Versi aturan immutable** (update membuat versi baru)  
✅ **Seed awal aman** (hanya metadata, status UNVERIFIED)  
✅ **Validasi publikasi** (gagal jika ada aturan mengikat UNVERIFIED)  
✅ **Dua tingkat snapshot** (policy pack + decision reference)  
✅ **Actor nullable sementara** dengan actorType
