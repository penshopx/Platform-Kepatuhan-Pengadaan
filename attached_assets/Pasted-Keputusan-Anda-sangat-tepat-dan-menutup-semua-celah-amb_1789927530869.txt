## 📋 IMPLIKASI TEKNIS PER KEPUTUSAN

### 1. Clerk sebagai Sumber Identitas

**Implikasi arsitektural:**
- Middleware Clerk wajib di semua endpoint `POST/PATCH/DELETE` pada `/api/admin/*` dan `/api/rules/evaluate`
- Session variables PostgreSQL harus di-set dari Clerk user, bukan dari request body

**Pola implementasi di API server:**

```typescript
// artifacts/api-server/src/middleware/clerk-audit.ts
import { getAuth } from '@clerk/express';
import { db } from '@procurement/db';
import { sql } from 'drizzle-orm';

export async function setAuditContext(req, res, next) {
  const { userId, orgId } = getAuth(req);
  
  if (!userId) {
    return res.status(401).json({ error: 'Authentication required' });
  }

  const reason = req.body?.reason || req.headers['x-audit-reason'];
  if (!reason && ['PATCH', 'DELETE', 'POST'].includes(req.method)) {
    // Reason wajib untuk operasi mutasi kritis
    const criticalPaths = ['/admin/legal-sources', '/admin/business-rules', 
                           '/admin/policy-packs', '/rules/evaluate'];
    if (criticalPaths.some(p => req.path.startsWith(p))) {
      return res.status(400).json({ error: 'X-Audit-Reason header or body.reason required' });
    }
  }

  // Set PostgreSQL session variables
  await db.execute(sql`
    SELECT set_audit_context(
      ${userId}::uuid,
      'USER',
      ${req.headers['x-actor-role'] || 'ADMIN'}::text,
      ${reason || ''}::text
    )
  `);

  req.auditContext = { userId, orgId, reason };
  next();
}
```

**Konsekuensi pada skema:**
- Kolom `verifiedBy`, `publishedBy`, `evaluatedBy`, `actorId` sekarang **de facto NOT NULL** di level aplikasi (meskipun DB masih nullable untuk kompatibilitas data seed lama)
- Tambahkan validasi di layer service: tolak operasi jika `userId` tidak tersedia

export async function evaluateWithContext(packId: string, context: EvaluationContext) {
  // 1. Ambil pack
  const pack = await db.query.policyPacks.findFirst({
    where: eq(policyPacks.id, packId)
  });

  if (!pack || pack.status !== 'PUBLISHED') {
    throw new Error('Only PUBLISHED policy packs can be evaluated');
  }

  // 2. Ambil RULES + SNAPSHOT, bukan business_rules live
  const packRules = await db.select({
    packRule: policyPackRules,
    // Snapshot sudah mengandung semua yang dibutuhkan
  }).from(policyPackRules)
    .where(eq(policyPackRules.policyPackId, packId));

  for (const pr of packRules) {
    if (!pr.packRule.snapshot) {
      throw new Error(`Snapshot missing for rule ${pr.packRule.businessRuleId}`);
    }
  }

  // 4. Evaluasi menggunakan snapshot, bukan live data

}
